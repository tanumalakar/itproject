// Firestore rules for a public portfolio with a protected admin workflow.
// IMPORTANT: These rules allow public writes and are NOT recommended for production.
// Before public deployment, add Firebase Authentication and restrict writes to admins.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /works/{workId} {
      allow read: if true;
      allow write: if true;
    }
  }
}
